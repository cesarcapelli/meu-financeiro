                                            {posted ? "Pago / Lançado ✅" : "Pendente"}

"50%" : tpl.splitRatio.split("/")[0] + "%"}) · {partnerName} reembolsa <span className="font-semibold text-green-500 font-mono">{fmt(partnerShare)}</span>
